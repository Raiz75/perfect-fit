# Admin Dashboard PDF Report Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a "Generate Report" button to the admin dashboard that exports filtered dashboard data (stat cards, server-side chart images, data tables) as a Dompdf PDF matching the legacy report format.

**Architecture:** Two new services (ChartImageService for pChart rendering, ReportController for filter/aggregate/PDF generation), one Dompdf Blade layout, and a small frontend JS snippet to submit filter state. Data aggregation logic mirrors what `admin-dashboard.js`'s `buildCharts()` does — same bucket boundaries, same labels, same order.

**Tech Stack:** Laravel 12, PHP 8.2+, Dompdf (`barryvdh/laravel-dompdf`), pChart 4.0 (`cpchart/cpchart`), Blade, Chart.js color palette

## Global Constraints

- All filter params must match `DashboardController::getData()` exactly (field names, enum values, bucket boundaries)
- Chart colors must match `admin-dashboard.js` exactly (listed in spec)
- Seeder IDs are frozen — do not create/modify seeders
- PDF filename format: `{ChurchName}_Dashboard_Report.pdf`
- Footer must appear on every page: `Generated by: {adminEmail}`, `Generated on: {YYYY-MM-DD}`, `Page X of Y`
- When no data matches filters, still generate PDF with "No records found" message

- [ ] **Step 1: Create the service class**

```php
<?php

namespace App\Services;

use CpChart\Data;
use CpChart\Image;
use CpChart\Chart\Pie;
use CpChart\Chart\Bar;
use Illuminate\Support\Facades\Storage;

class ChartImageService
{
    private string $tempDir = 'report-charts';

    /**
     * Render a pie chart and return the temp storage path.
     *
     * @param array $labels
     * @param array $data      Numeric values
     * @param array $colors    Hex color strings, e.g. ['#E6194B', '#F58231']
     * @param string $title
     * @return string Relative path in storage (e.g. "report-charts/gender.png")
     */
    public function renderPie(array $labels, array $data, array $colors, string $title): string
    {
        $pieData = new Data();
        foreach ($data as $i => $value) {
            $pieData->addPoints($value, 'Series1');
        }
        $pieData->addPoints($labels, 'Labels');
        $pieData->setAbscissa('Labels');

        $image = new Image(500, 300, $pieData);
        $image->setGraphArea(20, 40, 480, 260);
        $image->drawFilledRectangle(0, 0, 500, 300, [
            'R' => 255, 'G' => 255, 'B' => 255
        ]);
        $image->drawText(250, 25, $title, [
            'R' => 140, 'G' => 82, 'B' => 255, 'FontSize' => 14, 'Align' => 'TEXT_ALIGN_TOPMIDDLE'
        ]);

        $pie = new Pie($image, $pieData);
        $pie->draw2DPie(250, 170, [
            'Radius' => 100,
            'DrawLabels' => true,
            'LabelStacked' => true,
            'Border' => true,
        ]);

        return $this->saveImage($image);
    }

    /**
     * Render a doughnut chart and return the temp storage path.
     */
    public function renderDoughnut(array $labels, array $data, array $colors, string $title): string
    {
        $doughnutData = new Data();
        foreach ($data as $i => $value) {
            $doughnutData->addPoints($value, 'Series1');
        }
        $doughnutData->addPoints($labels, 'Labels');
        $doughnutData->setAbscissa('Labels');

        $image = new Image(500, 300, $doughnutData);
        $image->setGraphArea(20, 40, 480, 260);
        $image->drawFilledRectangle(0, 0, 500, 300, [
            'R' => 255, 'G' => 255, 'B' => 255
        ]);
        $image->drawText(250, 25, $title, [
            'R' => 140, 'G' => 82, 'B' => 255, 'FontSize' => 14, 'Align' => 'TEXT_ALIGN_TOPMIDDLE'
        ]);

        $pie = new Pie($image, $doughnutData);
        $pie->draw2DRing(250, 170, [
            'Radius' => 100,
            'DrawLabels' => true,
            'LabelStacked' => true,
            'Border' => true,
        ]);

        return $this->saveImage($image);
    }

    /**
     * Render a bar chart and return the temp storage path.
     */
    public function renderBar(array $labels, array $data, array $colors, string $title): string
    {
        $barData = new Data();
        $barData->addPoints($data, 'Series1');
        $barData->addPoints($labels, 'Labels');
        $barData->setAbscissa('Labels');

        $image = new Image(600, 350, $barData);
        $image->setGraphArea(60, 50, 570, 300);
        $image->drawFilledRectangle(0, 0, 600, 350, [
            'R' => 255, 'G' => 255, 'B' => 255
        ]);
        $image->drawText(300, 30, $title, [
            'R' => 140, 'G' => 82, 'B' => 255, 'FontSize' => 14, 'Align' => 'TEXT_ALIGN_TOPMIDDLE'
        ]);
        $image->drawScale($barData, [
            'Mode' => SCALE_MODE_START0,
            'GridR' => 200, 'GridG' => 200, 'GridB' => 200,
            'DrawSubTicks' => false,
            'LabelRotation' => 45,
        ]);

        $bar = new Bar($image, $barData);
        $bar->drawBarChart([
            'Surrounding' => 15,
            'OverrideColors' => $this->hexColorsToCpChart($colors),
        ]);

        return $this->saveImage($image);
    }

    /**
     * Convert hex color array to pChart color array.
     */
    private function hexColorsToCpChart(array $hexColors): array
    {
        $result = [];
        foreach ($hexColors as $hex) {
            $hex = ltrim($hex, '#');
            $result[] = [
                'R' => hexdec(substr($hex, 0, 2)),
                'G' => hexdec(substr($hex, 2, 2)),
                'B' => hexdec(substr($hex, 4, 2)),
            ];
        }
        return $result;
    }

    /**
     * Save the Image object to a temp PNG and return the relative path.
     */
    private function saveImage(Image $image): string
    {
        $filename = uniqid('chart_', true) . '.png';
        $relativePath = $this->tempDir . '/' . $filename;

        Storage::makeDirectory($this->tempDir);
        $image->render(Storage::path($relativePath));

        return $relativePath;
    }

    /**
     * Clean up all temp chart images for this request.
     */
    public function cleanup(): void
    {
        Storage::deleteDirectory($this->tempDir);
    }
}
```

```php
<?php

namespace App\Http\Controllers\Admin;

use App\Http\Controllers\Controller;
use App\Http\Requests\Admin\GenerateReportRequest;
use App\Models\UserReport;
use App\Services\ChartImageService;
use Barryvdh\DomPDF\Facade\Pdf;
use Illuminate\Support\Facades\Auth;

class ReportController extends Controller
{
    const MINISTRIES = [
        'Worship (Singing)', 'Worship (Dancing)', 'Worship (Instrument)', 'Prayer',
        'Preaching', 'Discipleship', 'Youth', 'Young Adults', "Men's", "Women's",
        'Family Or Couples', 'Ushering', 'Administrative', 'Finance', 'Marshal',
        'Facilities Maintenance', 'Evangelism', 'Missions', 'Community Service',
        'Visitation', 'Production Tech', 'Creative & Social Media', 'Counseling',
        'Healing & Deliverance', 'Funeral', 'Addiction Recovery', 'Special Needs',
        'Seniors', 'Single Adults',
    ];

    const GENDER_MAP = [1 => 'Male', 2 => 'Female'];
    const MARITAL_MAP = [1 => 'Single', 2 => 'Married'];
    const BAPTIZED_MAP = [1 => 'Yes', 2 => 'No'];
    const FAITH_MAP = [1 => '1+ Week', 2 => '6+ Months', 3 => '1+ Year', 4 => '2+ Years'];
    const FAITH_ORDER = ['1+ Week', '6+ Months', '1+ Year', '2+ Years'];
    const AGE_ORDER = ['Under 18', '18-25', '26-35', '36-50', '51+'];
    const SKILL_ORDER = ['music', 'technology', 'writing', 'technical', 'speaking', 'accounting', 'mentoring', 'bible_knowledge'];
    const SKILL_LABELS = ['Music', 'Technology', 'Writing', 'Technical', 'Speaking', 'Accounting', 'Mentoring', 'Bible Knowledge'];

    public function generate(GenerateReportRequest $request, ChartImageService $chartService)
    {
        $user = Auth::user();
        $query = UserReport::where('church_code', $user->church_code);

        // Apply filters (same logic as DashboardController::getData)
        if ($search = $request->input('search')) {
            $query->where(function ($q) use ($search) {
                $q->where('name', 'like', "%{$search}%")
                    ->orWhere('email', 'like', "%{$search}%");
            });
        }
        if ($startDate = $request->input('startDate')) {
            $query->whereDate('time_of_submission', '>=', $startDate);
        }
        if ($endDate = $request->input('endDate')) {
            $query->whereDate('time_of_submission', '<=', $endDate);
        }
        if ($gender = $request->input('gender')) {
            $query->where('gender', $gender);
        }
        if ($marital = $request->input('marital')) {
            $query->where('marital_status', $marital);
        }
        if ($baptized = $request->input('baptized')) {
            $query->where('baptized', $baptized);
        }
        if ($faith = $request->input('faith')) {
            $query->where('time_in_faith', $faith);
        }
        if ($age = $request->input('age')) {
            $query->where('age', $age);
        }
        if ($skills = $request->input('skills')) {
            $skillList = explode(',', $skills);
            foreach ($skillList as $skill) {
                $skill = trim($skill);
                if (in_array($skill, self::SKILL_ORDER)) {
                    $query->where($skill, 1);
                }
            }
        }
        if ($ministries = $request->input('ministries')) {
            $ministryList = explode(',', $ministries);
            $query->where(function ($q) use ($ministryList) {
                foreach ($ministryList as $ministry) {
                    $ministry = trim($ministry);
                    $q->orWhere('eligible_ministry', 'like', "%{$ministry}%");
                }
            });
        }

        $reports = $query->orderBy('time_of_submission', 'desc')->get();
        $totalTakers = $reports->count();

        // Build chart data (matching admin-dashboard.js buildCharts())
        $chartData = $this->buildChartData($reports);

        // Generate chart images
        $chartImages = $this->generateChartImages($chartService, $chartData);

        // Build filters list for the "Filters Applied" section
        $filters = $this->buildFiltersList($request);

        // Prepare table rows
        $tableRows = $reports->map(function ($r) {
            $skills = array_filter(self::SKILL_LABELS, fn($label, $i) => $r->{self::SKILL_ORDER[$i]} == 1, ARRAY_FILTER_USE_BOTH);
            $ministries = $r->eligible_ministry
                ? array_map('trim', explode(',', $r->eligible_ministry))
                : [];

            return [
                'name' => $r->name ?? '—',
                'email' => $r->email ?? '—',
                'skills' => $skills,
                'eligible_ministries' => $ministries,
                'gender' => self::GENDER_MAP[$r->gender] ?? '—',
                'age' => $r->age ?? '—',
                'marital' => self::MARITAL_MAP[$r->marital_status] ?? '—',
                'baptized' => self::BAPTIZED_MAP[$r->baptized] ?? '—',
                'faith' => self::FAITH_MAP[$r->time_in_faith] ?? '—',
                'date' => $r->time_of_submission ?? '—',
            ];
        });

        $pdf = Pdf::loadView('admin.reports.dashboard-pdf', [
            'churchName' => $user->church_name ?? 'PERFIT',
            'adminEmail' => $user->email,
            'generatedAt' => now()->format('Y-m-d H:i'),
            'startDate' => $request->input('startDate'),
            'endDate' => $request->input('endDate'),
            'filters' => $filters,
            'totalTakers' => $totalTakers,
            'chartImages' => $chartImages,
            'chartData' => $chartData,
            'tableRows' => $tableRows,
            'genderMap' => self::GENDER_MAP,
            'maritalMap' => self::MARITAL_MAP,
            'baptizedMap' => self::BAPTIZED_MAP,
            'faithMap' => self::FAITH_MAP,
        ]);

        $filename = preg_replace('/[^a-zA-Z0-9_-]/', '_', $user->church_name ?? 'PERFIT') . '_Dashboard_Report.pdf';

        // Clean up temp chart images
        $chartService->cleanup();

        return $pdf->download($filename);
    }

    private function buildChartData($reports): array
    {
        $data = [
            'gender' => [],
            'age' => array_fill_keys(self::AGE_ORDER, 0),
            'baptized' => ['Yes' => 0, 'No' => 0],
            'faith' => array_fill_keys(self::FAITH_ORDER, 0),
            'skills' => array_fill_keys(self::SKILL_LABELS, 0),
            'ministry' => [],
            'marital' => ['Single' => 0, 'Married' => 0],
        ];

        foreach ($reports as $r) {
            if ($r->gender && isset(self::GENDER_MAP[$r->gender])) {
                $label = self::GENDER_MAP[$r->gender];
                $data['gender'][$label] = ($data['gender'][$label] ?? 0) + 1;
            }
            if ($r->age) {
                $bucket = $this->bucketAge($r->age);
                $data['age'][$bucket] = ($data['age'][$bucket] ?? 0) + 1;
            }
            if ($r->baptized && isset(self::BAPTIZED_MAP[$r->baptized])) {
                $label = self::BAPTIZED_MAP[$r->baptized];
                if (isset($data['baptized'][$label])) {
                    $data['baptized'][$label]++;
                }
            }
            if ($r->time_in_faith && isset(self::FAITH_MAP[$r->time_in_faith])) {
                $label = self::FAITH_MAP[$r->time_in_faith];
                if (isset($data['faith'][$label])) {
                    $data['faith'][$label]++;
                }
            }
            foreach (self::SKILL_ORDER as $i => $skill) {
                if ($r->{$skill} == 1) {
                    $data['skills'][self::SKILL_LABELS[$i]]++;
                }
            }
            if ($r->eligible_ministry) {
                $ministries = array_map('trim', explode(',', $r->eligible_ministry));
                foreach ($ministries as $m) {
                    if ($m) {
                        $data['ministry'][$m] = ($data['ministry'][$m] ?? 0) + 1;
                    }
                }
            }
            if ($r->marital_status && isset(self::MARITAL_MAP[$r->marital_status])) {
                $label = self::MARITAL_MAP[$r->marital_status];
                if (isset($data['marital'][$label])) {
                    $data['marital'][$label]++;
                }
            }
        }

        return $data;
    }

    private function bucketAge(int $age): string
    {
        if ($age < 18) return 'Under 18';
        if ($age <= 25) return '18-25';
        if ($age <= 35) return '26-35';
        if ($age <= 50) return '36-50';
        return '51+';
    }

    private function generateChartImages(ChartImageService $service, array $chartData): array
    {
        $images = [];
        $palettes = $this->getChartPalettes();

        // Gender (pie)
        if (!empty($chartData['gender'])) {
            $images['gender'] = $service->renderPie(
                array_keys($chartData['gender']),
                array_values($chartData['gender']),
                $palettes['gender'],
                'Gender Distribution'
            );
        }

        // Age (pie)
        $ageData = array_values($chartData['age']);
        if (array_sum($ageData) > 0) {
            $images['age'] = $service->renderPie(
                self::AGE_ORDER,
                $ageData,
                $palettes['age'],
                'Age Groups'
            );
        }

        // Baptized (doughnut)
        $baptizedData = array_values($chartData['baptized']);
        if (array_sum($baptizedData) > 0) {
            $images['baptized'] = $service->renderDoughnut(
                array_keys($chartData['baptized']),
                $baptizedData,
                $palettes['baptized'],
                'Baptized Status'
            );
        }

        // Faith (bar)
        $faithData = array_values($chartData['faith']);
        if (array_sum($faithData) > 0) {
            $images['faith'] = $service->renderBar(
                self::FAITH_ORDER,
                $faithData,
                $palettes['faith'],
                'Time in Faith'
            );
        }

        // Skills (bar)
        $skillData = array_values($chartData['skills']);
        if (array_sum($skillData) > 0) {
            $images['skills'] = $service->renderBar(
                self::SKILL_LABELS,
                $skillData,
                $palettes['skills'],
                'Skills Breakdown'
            );
        }

        // Ministry (bar)
        if (!empty($chartData['ministry'])) {
            $images['ministry'] = $service->renderBar(
                array_keys($chartData['ministry']),
                array_values($chartData['ministry']),
                $this->hslColors(count($chartData['ministry'])),
                'Ministry Eligibility'
            );
        }

        // Marital (bar)
        $maritalData = array_values($chartData['marital']);
        if (array_sum($maritalData) > 0) {
            $images['marital'] = $service->renderBar(
                array_keys($chartData['marital']),
                $maritalData,
                $palettes['marital'],
                'Marital Status'
            );
        }

        return $images;
    }

    private function getChartPalettes(): array
    {
        return [
            'gender' => ['#E6194B', '#F58231'],
            'age' => ['#9A6324', '#FFD8B1', '#808000', '#AAFFC3', '#FF7F50'],
            'baptized' => ['#3CB44B', '#808080'],
            'faith' => ['#FFE119', '#BFEF45', '#3CB44B', '#46F0F0'],
            'skills' => ['#E6194B', '#F58231', '#FFE119', '#BFEF45', '#3CB44B', '#46F0F0', '#4363D8', '#911EB4'],
            'marital' => ['#800000', '#808080'],
        ];
    }

    private function hslColors(int $count): array
    {
        $colors = [];
        for ($i = 0; $i < $count; $i++) {
            $hue = ($i * 360) / $count;
            $colors[] = $this->hslToHex($hue, 70, 60);
        }
        return $colors;
    }

    private function hslToHex(float $h, int $s, int $l): string
    {
        $h /= 360;
        $s /= 100;
        $l /= 100;

        $c = (1 - abs(2 * $l - 1)) * $s;
        $x = $c * (1 - abs(fmod($h * 6, 2) - 1));
        $m = $l - $c / 2;

        $rgb = match (floor($h * 6)) {
            0 => [$c, $x, 0],
            1 => [$x, $c, 0],
            2 => [0, $c, $x],
            3 => [0, $x, $c],
            4 => [$x, 0, $c],
            5 => [$c, 0, $x],
            default => [0, 0, 0],
        };

        return sprintf('#%02x%02x%02x',
            round(($rgb[0] + $m) * 255),
            round(($rgb[1] + $m) * 255),
            round(($rgb[2] + $m) * 255)
        );
    }

    private function buildFiltersList(GenerateReportRequest $request): array
    {
        $filters = [
            ['title' => 'Name or Email', 'value' => $request->input('search', 'N/A')],
            ['title' => 'Age', 'value' => $request->input('age', 'N/A')],
        ];

        $genderVal = $request->input('gender');
        $filters[] = ['title' => 'Gender', 'value' => $genderVal ? (self::GENDER_MAP[(int)$genderVal] ?? 'N/A') : 'N/A'];

        $maritalVal = $request->input('marital');
        $filters[] = ['title' => 'Marital Status', 'value' => $maritalVal ? (self::MARITAL_MAP[(int)$maritalVal] ?? 'N/A') : 'N/A'];

        $baptizedVal = $request->input('baptized');
        $filters[] = ['title' => 'Baptized Status', 'value' => $baptizedVal ? (self::BAPTIZED_MAP[(int)$baptizedVal] ?? 'N/A') : 'N/A'];

        $faithVal = $request->input('faith');
        $filters[] = ['title' => 'Time in Faith', 'value' => $faithVal ? (self::FAITH_MAP[(int)$faithVal] ?? 'N/A') : 'N/A'];

        $skills = $request->input('skills');
        $filters[] = ['title' => 'Skills', 'value' => $skills ? str_replace(',', ' | ', $skills) : 'N/A'];

        $ministries = $request->input('ministries');
        $filters[] = ['title' => 'Ministries', 'value' => $ministries ? str_replace(',', ' | ', $ministries) : 'N/A'];

        return $filters;
    }
}
```

- [ ] **Step 3: Create the report button JS**

`resources/js/report-button.js`:

```js
document.addEventListener('DOMContentLoaded', function () {
    const generateBtn = document.getElementById('generateReportBtn');
    if (!generateBtn) return;

    generateBtn.addEventListener('click', function () {
        const form = document.createElement('form');
        form.method = 'POST';
        form.action = '/admin/dashboard/report';

        // CSRF token
        const csrf = document.createElement('input');
        csrf.type = 'hidden';
        csrf.name = '_token';
        csrf.value = document.querySelector('meta[name="csrf-token"]')?.content || '';
        form.appendChild(csrf);

        // Collect filter values (same fields as loadDashboardData)
        const filterFields = {
            search: 'searchInput',
            startDate: 'startDate',
            endDate: 'endDate',
            gender: 'genderFilter',
            marital: 'maritalFilter',
            baptized: 'baptizedFilter',
            faith: 'faithFilter',
            age: 'ageFilter',
        };

        Object.entries(filterFields).forEach(([name, id]) => {
            const el = document.getElementById(id);
            if (el && el.value) {
                const input = document.createElement('input');
                input.type = 'hidden';
                input.name = name;
                input.value = el.value;
                form.appendChild(input);
            }
        });

        // Skills (checkboxes)
        const skills = Array.from(document.querySelectorAll('.skill-filter:checked')).map(cb => cb.value);
        if (skills.length) {
            const input = document.createElement('input');
            input.type = 'hidden';
            input.name = 'skills';
            input.value = skills.join(',');
            form.appendChild(input);
        }

        // Ministries (checkboxes)
        const ministries = Array.from(document.querySelectorAll('.ministry-filter:checked')).map(cb => cb.value);
        if (ministries.length) {
            const input = document.createElement('input');
            input.type = 'hidden';
            input.name = 'ministries';
            input.value = ministries.join(',');
            form.appendChild(input);
        }

        document.body.appendChild(form);
        form.submit();
    });
});
```
